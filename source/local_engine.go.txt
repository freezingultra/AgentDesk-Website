func localVirtualProvider(m LocalModel) Provider {
	id := "local:" + m.ID
	return Provider{ID: id, Name: "Local · " + m.Name, BaseURL: "local://" + m.ID, Model: m.ID, Models: []string{m.ID}, MaxNewTok: 4096}
}

func localModelByID(id string) (LocalModel, error) {
	s := loadLocalSettings()
	for _, m := range s.Models {
		if m.ID == id || m.File == id {
			return m, nil
		}
	}
	return LocalModel{}, fmt.Errorf("local model not found")
}

func localProviderFromID(id string) (Provider, error) {
	modelID := strings.TrimPrefix(id, "local:")
	m, err := localModelByID(modelID)
	if err != nil { return Provider{}, err }
	base, err := ensureLocalRuntime(m)
	if err != nil { return Provider{}, err }
	p := localVirtualProvider(m)
	p.BaseURL = base
	p.Model = m.ID
	return p, nil
}

func parseLocalProviderID(id string) bool { return strings.HasPrefix(strings.TrimSpace(id), "local:") }

func scanLocalHardware() (map[string]any, error) {
	cores := runtime.NumCPU()
	threads := runtime.NumCPU()
	gpuName, vram := "", 0.0
	hasGPU := false
	ramGB, ramSpeed := systemMemory()
	if runtime.GOOS == "windows" {
		if name, total, ok := scanNvidiaWindows(); ok {
			hasGPU, gpuName, vram = true, name, total
		}
		if v := powershellText("(Get-CimInstance Win32_Processor | Select-Object -First 1 -ExpandProperty NumberOfCores)"); v != "" { cores, _ = strconv.Atoi(strings.TrimSpace(v)) }
		if v := powershellText("(Get-CimInstance Win32_Processor | Select-Object -First 1 -ExpandProperty NumberOfLogicalProcessors)"); v != "" { threads, _ = strconv.Atoi(strings.TrimSpace(v)) }
	}
	score := benchmarkLocalCPU()
	tier := classifyLocalCPU(cores, score)
	s := loadLocalSettings()
	s.HasNvidiaGPU, s.GPUName, s.VRAMGB = hasGPU, gpuName, vram
	s.CPUCores, s.CPUThreads, s.CPUBenchmark, s.CPUTier = cores, threads, score, tier
	s.RAMGB, s.RAMSpeedMHz = ramGB, ramSpeed
	s.Models = detectExistingLocalModels()
	_ = saveLocalSettings(s)
	for _, m := range s.Models { upsertLocalProvider(m) }
	return map[string]any{
		"nvidia": hasGPU, "gpuName": gpuName, "vramGB": vram,
		"cpuCores": cores, "cpuThreads": threads, "cpuTier": tier, "cpuBenchmark": score,
		"ramGB": ramGB, "ramSpeedMHz": ramSpeed, "os": runtime.GOOS,
	}, nil
}

func powershellText(script string) string {
	cmd := exec.Command("powershell.exe", "-NoLogo", "-NoProfile", "-NonInteractive", "-ExecutionPolicy", "Bypass", "-Command", script)
	configureHiddenCommand(cmd)
	var out bytes.Buffer; cmd.Stdout = &out; cmd.Stderr = io.Discard
	if err := cmd.Run(); err != nil { return "" }
	return strings.TrimSpace(out.String())
}

func scanNvidiaWindows() (string, float64, bool) {
	if _, err := exec.LookPath("nvidia-smi.exe"); err != nil {
		if _, err = exec.LookPath("nvidia-smi"); err != nil { return "", 0, false }
	}
	cmd := exec.Command("nvidia-smi", "--query-gpu=name,memory.total", "--format=csv,noheader,nounits")
	configureHiddenCommand(cmd)
	b, err := cmd.Output()
	if err != nil { return "", 0, false }
	line := strings.TrimSpace(strings.Split(string(b), "\n")[0])
	parts := strings.Split(line, ",")
	if len(parts) < 2 { return "", 0, false }
	mb, err := strconv.ParseFloat(strings.TrimSpace(parts[1]), 64)
	if err != nil { return "", 0, false }
	return strings.TrimSpace(parts[0]), mb / 1024.0, true
}

func systemMemory() (float64, int) {
	if runtime.GOOS == "windows" {
		total := powershellText("[math]::Round((Get-CimInstance Win32_ComputerSystem).TotalPhysicalMemory / 1GB, 2)")
		speed := powershellText("$x=Get-CimInstance Win32_PhysicalMemory | Where-Object ConfiguredClockSpeed | Measure-Object ConfiguredClockSpeed -Average; [math]::Round($x.Average)")
		g, _ := strconv.ParseFloat(strings.TrimSpace(total), 64); mhz, _ := strconv.Atoi(strings.TrimSpace(speed))
		return g, mhz
	}
	if runtime.GOOS == "linux" {
		if b, err := os.ReadFile("/proc/meminfo"); err == nil {
			for _, line := range strings.Split(string(b), "\n") {
				if strings.HasPrefix(line, "MemTotal:") {
					f := strings.Fields(line); if len(f) >= 2 { kb, _ := strconv.ParseFloat(f[1], 64); return kb / 1024 / 1024, 0 }
				}
			}
		}
	}
	if runtime.GOOS == "darwin" {
		if out := commandOutput("sysctl", "-n", "hw.memsize"); out != "" { b, _ := strconv.ParseFloat(out, 64); return b / 1024 / 1024 / 1024, 0 }
	}
	return 0, 0
}

func commandOutput(name string, args ...string) string {
	cmd := exec.Command(name, args...); configureHiddenCommand(cmd); b, err := cmd.Output(); if err != nil { return "" }; return strings.TrimSpace(string(b))
}

func benchmarkLocalCPU() float64 {
	buf := make([]byte, 4096)
	for i := range buf { buf[i] = byte(i * 31) }
	start := time.Now(); loops := int64(0); var digest [32]byte
	for time.Since(start) < 350*time.Millisecond {
		digest = sha256.Sum256(buf); buf[0] ^= digest[0]; loops++
	}
	if loops <= 0 { return 0 }
	return float64(loops) / 1000.0
}

func classifyLocalCPU(cores int, score float64) string {
	if cores > 16 || score >= 700 { return "Workstation · Threadripper/Epyc class" }
	if cores >= 8 && score >= 250 { return "High end · Ryzen 7/Ryzen 9/I7/I9 class" }
	if cores >= 6 && score >= 120 { return "Mid range · Ryzen 5/Old-I7 class" }
	return "Low end · I3/Old-I5 class"
}

func normalizeHFRepo(raw string) (repo, file string, err error) {
	x := strings.TrimSpace(raw)
	if x == "" { return "", "", fmt.Errorf("model URL is required") }
	if !strings.Contains(x, "://") && strings.Count(x, "/") == 1 { return x, "", nil }
	u, e := url.Parse(x); if e != nil { return "", "", fmt.Errorf("invalid model URL") }
	if !strings.EqualFold(u.Hostname(), "huggingface.co") { return "", "", fmt.Errorf("model URL must point to huggingface.co") }
	parts := strings.Split(strings.Trim(u.Path, "/"), "/")
	for len(parts) > 0 && (parts[0] == "models" || parts[0] == "model") { parts = parts[1:] }
	if len(parts) < 2 { return "", "", fmt.Errorf("use a Hugging Face model URL such as https://huggingface.co/org/model") }
	repo = parts[0] + "/" + parts[1]
	for i := 2; i < len(parts); i++ {
		if parts[i] == "resolve" || parts[i] == "blob" { if i+2 < len(parts) { file = strings.Join(parts[i+2:], "/") }; break }
	}
	return repo, file, nil
}

func hfRequest(path, token string) (*http.Response, error) {
	req, err := http.NewRequest("GET", "https://huggingface.co"+path, nil); if err != nil { return nil, err }
	req.Header.Set("User-Agent", "AgentDesk/"+appVersion)
	if strings.TrimSpace(token) != "" { req.Header.Set("Authorization", "Bearer "+strings.TrimSpace(token)) }
	return (&http.Client{Timeout: 45*time.Second}).Do(req)
}

func inspectHFModel(rawURL, token string) (LocalModelInfo, bool, error) {
	repo, direct, err := normalizeHFRepo(rawURL); if err != nil { return LocalModelInfo{}, false, err }
	resp, err := hfRequest("/api/models/"+url.PathEscape(repo), token); if err != nil { return LocalModelInfo{}, false, err }
	defer resp.Body.Close()
	body, _ := io.ReadAll(io.LimitReader(resp.Body, 32<<20))
	if resp.StatusCode == 401 || resp.StatusCode == 403 { return LocalModelInfo{}, true, fmt.Errorf("Hugging Face requires authentication for this model") }
	if resp.StatusCode == 404 { return LocalModelInfo{}, false, fmt.Errorf("Hugging Face model not found: %s", repo) }
	if resp.StatusCode >= 400 { return LocalModelInfo{}, false, fmt.Errorf("Hugging Face HTTP %d: %s", resp.StatusCode, strings.TrimSpace(string(body))) }
	var raw struct { Gated any `json:"gated"`; Private bool `json:"private"`; Siblings []struct { RFilename string `json:"rfilename"`; Size int64 `json:"size"` } `json:"siblings"` }
	if err := json.Unmarshal(body, &raw); err != nil { return LocalModelInfo{}, false, err }
	gated := false; switch v := raw.Gated.(type) { case bool: gated=v; case string: gated=strings.EqualFold(v,"true") }
	files := make([]map[string]any,0,len(raw.Siblings))
	for _, f := range raw.Siblings { if strings.TrimSpace(f.RFilename)=="" { continue }; files=append(files,map[string]any{"name":f.RFilename,"size":f.Size}) }
	info := LocalModelInfo{RepoID:repo,Gated:gated,Files:files}
	if direct != "" { for _, f := range files { if f["name"] == direct { if !strings.HasSuffix(strings.ToLower(direct), ".gguf") { return info, false, fmt.Errorf("AgentDesk Local Engine currently requires a GGUF model file") }; info.Selected=[]map[string]any{f}; return info, false, nil } } }
	info.Selected = selectLocalGGUFs(files, loadLocalSettings(), false)
	if len(info.Selected)==0 { return info, false, fmt.Errorf("no GGUF model file was found. AgentDesk Local Engine currently runs GGUF models; other Hub formats need a compatible conversion.") }
	return info, false, nil
}

func selectLocalGGUFs(files []map[string]any, s LocalSettings, vision bool) []map[string]any {
	type candidate struct { m map[string]any; score int; size int64 }
	var cs []candidate
	for _, f := range files {
		n,_ := f["name"].(string); low:=strings.ToLower(n); if !strings.HasSuffix(low,".gguf") || strings.Contains(low,"mmproj") { continue }
		if vision && !strings.Contains(low,"vision") { continue }
		if !vision && strings.Contains(low,"vision") && !strings.Contains(low,"instruct") { continue }
		size := int64(0); switch v:=f["size"].(type){case float64:size=int64(v);case int64:size=v}
		score:=0
		for i,q:=range []string{"q4_k_m","q5_k_m","q4_k_s","q5_k_s","q6_k","q8_0","q3_k_m","q2_k"}{if strings.Contains(low,q){score+=100-i*5;break}}
		if strings.Contains(low,"instruct")||strings.Contains(low,"it"){score+=25}
		budget:=s.RAMGB*1024*1024*1024*0.70 + s.VRAMGB*1024*1024*1024*0.85
		if budget>0 && size>0 && float64(size)>budget {score-=1000}
		cs=append(cs,candidate{f,score,size})
	}
	sort.Slice(cs,func(i,j int)bool{if cs[i].score!=cs[j].score{return cs[i].score>cs[j].score};return cs[i].size<cs[j].size})
	if len(cs)==0{return nil}; return []map[string]any{cs[0].m}
}
