  try{
    if($('#terminalMode').checked&&!$('#terminalMode').dataset.confirmed){
      const ok=confirm('Allow AgentDesk to run shell commands on this computer for this chat?');
      if(!ok)$('#terminalMode').checked=false;else $('#terminalMode').dataset.confirmed='1';
    }
    const geo=weatherish(text)?await getBrowserLocation():null;
    const research=$('#researchMode').checked;
    const agent=$('#agentMode').checked||research||webish(text)||weatherish(text);
    const r=await fetch('/api/chat/stream',{
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({provider:p.id,model:currentModel(),messages:state.messages,maxNewTok:Number(p.maxNewTok)||4096,agent,research,allowTerminal:$('#terminalMode').checked,location:geo})
    });
    if(!r.ok)throw new Error(await r.text());
    const reader=r.body.getReader();
    const dec=new TextDecoder();
    let buf='',out='',bubble=null;
    const thinking=addThinkingMessage();
    for(;;){
      const {value,done}=await reader.read();
      if(done)break;
      buf+=dec.decode(value,{stream:true});
      const chunks=buf.split('\n\n');
      buf=chunks.pop()||'';
      for(const part of chunks){
        const line=part.split('\n').find(x=>x.startsWith('data: '));
        if(!line)continue;
        const ev=JSON.parse(line.slice(6));
        if(ev.type==='tool')addTool(ev.text);
        else if(ev.type==='market')renderMarketCard(ev.data);
        else if(ev.type==='sources')addChatSources(ev.sources||[]);
        else if(ev.type==='notice')addTool(ev.text||'Notice');
        else if(ev.type==='chunk'){
          hideThinking(thinking);
          out+=ev.text;
          if(!bubble)bubble=addMsg('AgentDesk','');
          bubble.innerHTML=md(out);
          bindCodeActions(bubble.parentElement);
          attachPackageButton(bubble.parentElement,out);
          $('#messages').scrollTop=$('#messages').scrollHeight;
        }else if(ev.type==='error'){ hideThinking(thinking); throw new Error(ev.error); }
      }
    }
    if(!out&&!bubble){hideThinking(thinking);throw new Error('Model returned no text');}
    hideThinking(thinking);
    hideActivity();
    state.messages.push({role:'assistant',content:out});
    await persistChat();
    renderChats();
  }catch(e){
    document.querySelector('.thinkingMsg')?.remove();
    hideActivity();
    addMsg('AgentDesk','Error: '+e.message);
    state.messages.push({role:'assistant',content:'Error: '+e.message});
    await persistChat();
  }
}
